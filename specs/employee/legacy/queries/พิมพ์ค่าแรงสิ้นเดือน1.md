```sql
SELECT [สำเนาของ ค่าแรง].ID, [สำเนาของ ค่าแรง].NAME, [สำเนาของ ค่าแรง].W, [สำเนาของ ค่าแรง].PAY_W, [สำเนาของ ค่าแรง].FROM_DATE, EMPLO.ID_CODE, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].F_D, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].T_D, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].MO, [สำเนาของ ค่าแรง].TO_DATE, [สำเนาของ ค่าแรง].TT_OT, EMPLO.DEPARTMENT, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].DEP
FROM ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน, EMPLO INNER JOIN [สำเนาของ ค่าแรง] ON EMPLO.ID = [สำเนาของ ค่าแรง].ID
GROUP BY [สำเนาของ ค่าแรง].ID, [สำเนาของ ค่าแรง].NAME, [สำเนาของ ค่าแรง].W, [สำเนาของ ค่าแรง].PAY_W, [สำเนาของ ค่าแรง].FROM_DATE, EMPLO.ID_CODE, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].F_D, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].T_D, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].MO, [สำเนาของ ค่าแรง].TO_DATE, [สำเนาของ ค่าแรง].TT_OT, EMPLO.DEPARTMENT, [ตั้งวันที่พิมพ์ค่าแรงสิ้นเดือน].DEP
```

## พิมพ์ค่าแรงสิ้นเดือนแผนกจัดซื้อ

- ชนิด: **SELECT** · แก้ล่าสุด 2021-05-12
- อ่านจาก: [พิมพ์ค่าแรงสิ้นเดือน3](<../queries/พิมพ์ค่าแรงสิ้นเดือน1.md#q-พิมพ์ค่าแรงสิ้นเดือน3>)
- ใช้โดย: คิวรี [พิมพ์ค่าแรงสิ้นเดือนแผนกจัดซื้อ1](<../queries/พิมพ์ค่าแรงสิ้นเดือน1.md#q-พิมพ์ค่าแรงสิ้นเดือนแผนกจัดซื้อ1>) (SQL); มาโคร [พิมพ์ค่าแรงสิ้นเดือน1](<../macros.md#m-พิมพ์ค่าแรงสิ้นเดือน1>) (OpenQuery)

```sql
SELECT [พิมพ์ค่าแรงสิ้นเดือน3].DEPARTMENT, Sum([พิมพ์ค่าแรงสิ้นเดือน3].PAY_W) AS SumOfPAY_W, [พิมพ์ค่าแรงสิ้นเดือน3].ID, [พิมพ์ค่าแรงสิ้นเดือน3].MO
FROM พิมพ์ค่าแรงสิ้นเดือน3
GROUP BY [พิมพ์ค่าแรงสิ้นเดือน3].DEPARTMENT, [พิมพ์ค่าแรงสิ้นเดือน3].ID, [พิมพ์ค่าแรงสิ้นเดือน3].MO
HAVING ((([พิมพ์ค่าแรงสิ้นเดือน3].ID) Like "F*"));
```
