```sql
SELECT EMPLO.ID, EMPLO.DEPARTMENT, [ลางานทั้งแผนก].FORM, [ลางานทั้งแผนก].TO, [ลางานทั้งแผนก].DEP, [ลางานทั้งแผนก].EMP_QTY, [ลางานทั้งแผนก].DATE_QTY, EMPLO.RESIGN, [ลางานทั้งแผนก].PAY_M, [ลางานทั้งแผนก].PAY_D, [ลางานทั้งแผนก].DATE_H, E_WORK.Date, E_WORK.LATE
FROM ลางานทั้งแผนก, EMPLO INNER JOIN E_WORK ON EMPLO.ID = E_WORK.ID
WHERE (((EMPLO.DEPARTMENT) Like [dep]) AND ((EMPLO.RESIGN) Is Null) AND ((E_WORK.Date)>=[form] And (E_WORK.Date)<=[to]) AND ((E_WORK.LATE) Is Not Null));
```
