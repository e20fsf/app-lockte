Option Compare Database



Private Sub Command1_Click()

On Error GoTo Err_Command1_Click



    Dim stDocName As String



    stDocName = "ลบข้อมูลปีที่แล้ว"

    DoCmd.RunMacro stDocName

End Sub

End Sub
